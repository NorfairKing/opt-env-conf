  switch: -h|--help

  Output version information: 0.0.0
  switch: --version

  Path to the configuration file
  option: --config-file FILE_PATH

  an object with a default and an example
  config:

      # BigConfig
      map: # required
        <key>:

      sub: # optional
        <string>

  default: [1,2,3]

  a number with a codec that can only decode
  config:
    decode-only: # or null
      <integer> # 64 bit unsigned integer
  default: 5
